# AgentVoice

Voice option for an agent.

## Example Usage

```typescript
import { AgentVoice } from "syllable-sdk/models/components";

let value: AgentVoice = {
  provider: "Google",
  displayName: "Fenrir (Spanish)",
  varName: "wavenet:female/en-US-Chirp3-HD-Achernar",
  gender: "female",
  model: "Standard",
  supportedLanguages: [
    {
      name: "English",
      code: "en-US",
    },
  ],
  deprecated: false,
  status: "active",
};
```
