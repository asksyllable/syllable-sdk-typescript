# InsightToolResponse

Response model for insight tool saves, carrying any non-blocking findings.

## Example Usage

```typescript
import { InsightToolResponse } from "syllable-sdk/models/components";

let value: InsightToolResponse = {
  name: "summary-tool",
  description: "This tool uses GPT4.1 to generate a summary of the call",
  version: 1,
  toolArguments: {
    "prompt":
      "Provide a concise, accurate summary of the conversation's key points, focusing on the user's goal and how the agent responded",
  },
  insightToolDefinitionId: 1,
  id: 1,
  insightToolDefinition: null,
  createdAt: new Date("2026-10-24T00:00:00Z"),
  updatedAt: new Date("2026-10-25T00:00:00Z"),
  lastUpdatedBy: "user@email.com",
};
```
